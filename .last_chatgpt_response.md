public enum SourceSnapshotError: Error, CustomStringConvertible {
    case invalidSource(String)
    case changedDuringAcquisition
    case insufficientSpace(required: UInt64, available: UInt64)
    case existingBlobCorrupted
    case copyIncomplete
    case cannotLockStore

    public var description: String {
        switch self {
        case .invalidSource(let message):
            return "Invalid source: \(message)"
        case .changedDuringAcquisition:
            return "Source changed during acquisition"
        case .insufficientSpace(let required, let available):
            return "Required \(required), available \(available) bytes"
        case .existingBlobCorrupted:
            return "Existing snapshot has invalid SHA-256"
        case .copyIncomplete:
            return "Snapshot verification failed"
        case .cannotLockStore:
            return "Cannot lock snapshot store"
        }
    }
}

// Candidate only; NOT an authoritative database commit.
public struct SourceSnapshotCandidate: Codable, Sendable {
    public let sourceRevisionID: UUID
    public let originalPath: String
    public let snapshotRelativePath: String
    public let snapshotSHA256: String
    public let byteSize: UInt64
    public let mode: String
    public let acquisitionTime: Date
    public let deduplicated: Bool
    public let state: String

    private enum CodingKeys: String, CodingKey {
        case sourceRevisionID = "source_revision_id"
        case originalPath = "original_path"
        case snapshotRelativePath = "snapshot_relative_path"
        case snapshotSHA256 = "snapshot_sha256"
        case byteSize = "byte_size"
        case mode
        case acquisitionTime = "acquisition_time"
        case deduplicated
        case state
    }
}

private struct SourceStat: Equatable {
    let device: UInt64
    let inode: UInt64
    let size: UInt64
    let mode: UInt32
    let mtimeSec: Int64
    let mtimeNsec: Int64
    let ctimeSec: Int64
    let ctimeNsec: Int64

    init(_ value: stat) throws {
        guard (value.st_mode & mode_t(S_IFMT)) == mode_t(S_IFREG),
              value.st_size > 0 else {
            throw SourceSnapshotError.invalidSource(
                "Expected non-empty regular file"
            )
        }

        device = UInt64(truncatingIfNeeded: value.st_dev)
        inode = UInt64(truncatingIfNeeded: value.st_ino)
        size = UInt64(value.st_size)
        mode = UInt32(value.st_mode)

        #if canImport(Darwin)
        mtimeSec = Int64(value.st_mtimespec.tv_sec)
        mtimeNsec = Int64(value.st_mtimespec.tv_nsec)
        ctimeSec = Int64(value.st_ctimespec.tv_sec)
        ctimeNsec = Int64(value.st_ctimespec.tv_nsec)
        #else
        mtimeSec = Int64(value.st_mtim.tv_sec)
        mtimeNsec = Int64(value.st_mtim.tv_nsec)
        ctimeSec = Int64(value.st_ctim.tv_sec)
        ctimeNsec = Int64(value.st_ctim.tv_nsec)
        #endif
    }
}

public enum SourceSnapshotStore {
    private static let chunkSize = 1024 * 1024

    // Caller must maintain security-scoped access.
    // Execute off the main actor.
    // Only ProjectStore may commit a canonical source revision.
    public static func acquire(
        sourceURL: URL,
        projectRoot: URL,
        reserveBytes: UInt64 = 64 * 1024 * 1024,
        availableBytes: (() throws -> UInt64)? = nil,
        progress: ((UInt64) throws -> Void)? = nil
    ) throws -> SourceSnapshotCandidate {
        let fm = FileManager.default
        let source = sourceURL.standardizedFileURL
        let root = projectRoot.standardizedFileURL

        guard source.isFileURL, root.isFileURL else {
            throw SourceSnapshotError.invalidSource(
                "Only local file URLs supported"
            )
        }

        let resolvedRoot = root.resolvingSymlinksInPath().path
        let resolvedSource = source.resolvingSymlinksInPath().path

        guard resolvedSource != resolvedRoot,
              !resolvedSource.hasPrefix(resolvedRoot + "/") else {
            throw SourceSnapshotError.invalidSource(
                "Source cannot be inside managed project"
            )
        }

        let sources = root.appendingPathComponent(
            "sources", isDirectory: true
        )
        let staging = sources.appendingPathComponent(
            ".staging", isDirectory: true
        )
        let blobs = sources.appendingPathComponent(
            "blobs", isDirectory: true
        )

        try fm.createDirectory(
            at: staging, withIntermediateDirectories: true
        )
        try fm.createDirectory(
            at: blobs, withIntermediateDirectories: true
        )

        for directory in [sources, staging, blobs] {
            guard directory.resolvingSymlinksInPath()
                .path.hasPrefix(resolvedRoot + "/") else {
                throw SourceSnapshotError.invalidSource(
                    "Store resolves outside project"
                )
            }
        }

        let lockURL = sources.appendingPathComponent(".acquire.lock")
        let lockFD = lockURL.path.withCString {
            open($0, O_RDWR | O_CREAT, mode_t(0o600))
        }

        guard lockFD >= 0 else {
            throw SourceSnapshotError.cannotLockStore
        }
        defer { _ = close(lockFD) }

        guard flock(lockFD, LOCK_EX) == 0 else {
            throw SourceSnapshotError.cannotLockStore
        }
        defer { _ = flock(lockFD, LOCK_UN) }

        // Keep one open file descriptor throughout acquisition.
        let input = try FileHandle(forReadingFrom: source)
        defer { try? input.close() }

        let original = try descriptorStat(input.fileDescriptor)

        let freeBytes: UInt64
        if let availableBytes {
            freeBytes = try availableBytes()
        } else {
            let attributes = try fm.attributesOfFileSystem(
                forPath: sources.path
            )
            guard let free = attributes[.systemFreeSize] as? NSNumber else {
                throw SourceSnapshotError.invalidSource(
                    "Cannot establish available disk space"
                )
            }
            freeBytes = free.uint64Value
        }

        let (required, overflow) =
            original.size.addingReportingOverflow(reserveBytes)

        guard !overflow else {
            throw SourceSnapshotError.invalidSource("File too large")
        }
        guard freeBytes >= required else {
            throw SourceSnapshotError.insufficientSpace(
                required: required,
                available: freeBytes
            )
        }

        let temporary = staging.appendingPathComponent(
            "\(UUID().uuidString).part"
        )
        guard fm.createFile(
            atPath: temporary.path, contents: nil
        ) else {
            throw SourceSnapshotError.invalidSource(
                "Cannot create staging file"
            )
        }
        defer { try? fm.removeItem(at: temporary) }

        var copied: UInt64 = 0
        let copiedDigest: String

        do {
            let output = try FileHandle(forWritingTo: temporary)
            defer { try? output.close() }

            var digest = Digest256()
            try input.seek(toOffset: 0)

            while let data = try input.read(upToCount: chunkSize),
                  !data.isEmpty {
                digest.update(data)
                try output.write(contentsOf: data)
                copied += UInt64(data.count)
                try progress?(copied)
            }

            try output.synchronize()
            copiedDigest = digest.finish()
        }

        guard copied == original.size else {
            throw SourceSnapshotError.changedDuringAcquisition
        }

        guard try descriptorStat(input.fileDescriptor) == original else {
            throw SourceSnapshotError.changedDuringAcquisition
        }

        let rereadDigest = try hashHandle(input)

        guard rereadDigest == copiedDigest,
              try descriptorStat(input.fileDescriptor) == original,
              try pathStat(source.path) == original else {
            throw SourceSnapshotError.changedDuringAcquisition
        }

        guard try hashURL(temporary) == copiedDigest else {
            throw SourceSnapshotError.copyIncomplete
        }

        let shard = blobs.appendingPathComponent(
            String(copiedDigest.prefix(2)),
            isDirectory: true
        )
        try fm.createDirectory(
            at: shard, withIntermediateDirectories: true
        )

        let canonical = shard.appendingPathComponent(copiedDigest)
        let deduplicated: Bool

        if fm.fileExists(atPath: canonical.path) {
            guard try hashURL(canonical) == copiedDigest else {
                throw SourceSnapshotError.existingBlobCorrupted
            }
            deduplicated = true
        } else {
            try fm.moveItem(at: temporary, to: canonical)
            try syncDirectory(shard)
            deduplicated = false
        }

        guard try hashURL(canonical) == copiedDigest else {
            throw SourceSnapshotError.existingBlobCorrupted
        }

        let candidate = SourceSnapshotCandidate(
            sourceRevisionID: UUID(),
            originalPath: source.path,
            snapshotRelativePath:
                "sources/blobs/\(copiedDigest.prefix(2))/\(copiedDigest)",
            snapshotSHA256: copiedDigest,
            byteSize: copied,
            mode: "full_copy_verified",
            acquisitionTime: Date(),
            deduplicated: deduplicated,
            state: "VERIFIED_NOT_COMMITTED"
        )

        let manifestDir = sources.appendingPathComponent(
            "manifests", isDirectory: true
        )
        try fm.createDirectory(
            at: manifestDir, withIntermediateDirectories: true
        )

        let manifestTemp = manifestDir.appendingPathComponent(
            ".\(candidate.sourceRevisionID.uuidString).part"
        )
        let manifestURL = manifestDir.appendingPathComponent(
            "\(candidate.sourceRevisionID.uuidString).json"
        )
        defer { try? fm.removeItem(at: manifestTemp) }

        let encoder = JSONEncoder()
        encoder.dateEncodingStrategy = .iso8601
        encoder.outputFormatting = [.prettyPrinted, .sortedKeys]

        let manifest = try encoder.encode(candidate)
        guard fm.createFile(
            atPath: manifestTemp.path,
            contents: manifest
        ) else {
            throw SourceSnapshotError.invalidSource(
                "Cannot write import manifest"
            )
        }

        let manifestHandle = try FileHandle(
            forWritingTo: manifestTemp
        )
        try manifestHandle.synchronize()
        try manifestHandle.close()

        try fm.moveItem(at: manifestTemp, to: manifestURL)
        try syncDirectory(manifestDir)

        return candidate
    }

    @discardableResult
    public static func reconcileAbandonedStaging(
        projectRoot: URL,
        olderThan seconds: TimeInterval = 300
    ) throws -> Int {
        let fm = FileManager.default
        let sources = projectRoot.standardizedFileURL
            .appendingPathComponent("sources", isDirectory: true)
        let staging = sources.appendingPathComponent(
            ".staging", isDirectory: true
        )

        try fm.createDirectory(
            at: staging, withIntermediateDirectories: true
        )

        let lockURL = sources.appendingPathComponent(".acquire.lock")
        let lockFD = lockURL.path.withCString {
            open($0, O_RDWR | O_CREAT, mode_t(0o600))
        }

        guard lockFD >= 0 else {
            throw SourceSnapshotError.cannotLockStore
        }
        defer { _ = close(lockFD) }

        guard flock(lockFD, LOCK_EX) == 0 else {
            throw SourceSnapshotError.cannotLockStore
        }
        defer { _ = flock(lockFD, LOCK_UN) }

        let manifests = sources.appendingPathComponent(
            "manifests", isDirectory: true
        )
        try fm.createDirectory(
            at: manifests, withIntermediateDirectories: true
        )

        let cutoff = Date().addingTimeInterval(-max(0, seconds))
        var removed = 0

        for (directory, prefix) in [
            (staging, ""),
            (manifests, ".")
        ] {
            let urls = try fm.contentsOfDirectory(
                at: directory,
                includingPropertiesForKeys: nil
            )

            for url in urls {
                let name = url.lastPathComponent
                guard name.hasPrefix(prefix),
                      name.hasSuffix(".part") else { continue }

                let id = String(
                    name.dropFirst(prefix.count).dropLast(5)
                )
                guard UUID(uuidString: id) != nil else {
                    continue
                }

                let attrs = try fm.attributesOfItem(
                    atPath: url.path
                )
                guard attrs[.type] as? FileAttributeType == .typeRegular,
                      let modified = attrs[.modificationDate] as? Date,
                      modified <= cutoff else {
                    continue
                }

                try fm.removeItem(at: url)
                removed += 1
            }
        }

        return removed
    }

    public static func verify(
        _ candidate: SourceSnapshotCandidate,
        projectRoot: URL
    ) throws -> Bool {
        guard candidate.snapshotSHA256.count == 64,
              candidate.snapshotSHA256.allSatisfy({ $0.isHexDigit }),
              candidate.snapshotRelativePath ==
                "sources/blobs/\(candidate.snapshotSHA256.prefix(2))/\(candidate.snapshotSHA256)" else {
            return false
        }

        let root = projectRoot.standardizedFileURL
        let url = root.appendingPathComponent(
            candidate.snapshotRelativePath
        )

        guard url.resolvingSymlinksInPath().path.hasPrefix(
            root.resolvingSymlinksInPath().path + "/"
        ) else {
            return false
        }

        guard FileManager.default.fileExists(atPath: url.path) else {
            return false
        }

        let attrs = try FileManager.default.attributesOfItem(
            atPath: url.path
        )
        guard let size = attrs[.size] as? NSNumber,
              size.uint64Value == candidate.byteSize else {
            return false
        }

        return try hashURL(url) == candidate.snapshotSHA256
    }

    private static func descriptorStat(
        _ fd: Int32
    ) throws -> SourceStat {
        var result = stat()
        guard fstat(fd, &result) == 0 else {
            throw SourceSnapshotError.invalidSource(
                "Cannot inspect source descriptor"
            )
        }
        return try SourceStat(result)
    }

    private static func pathStat(
        _ path: String
    ) throws -> SourceStat {
        var result = stat()
        guard path.withCString({
            stat($0, &result)
        }) == 0 else {
            throw SourceSnapshotError.changedDuringAcquisition
        }
        return try SourceStat(result)
    }

    private static func hashHandle(
        _ handle: FileHandle
    ) throws -> String {
        try handle.seek(toOffset: 0)
        var digest = Digest256()

        while let data = try handle.read(upToCount: chunkSize),
              !data.isEmpty {
            digest.update(data)
        }

        return digest.finish()
    }

    private static func hashURL(
        _ url: URL
    ) throws -> String {
        let handle = try FileHandle(forReadingFrom: url)
        defer { try? handle.close() }
        return try hashHandle(handle)
    }

    private static func syncDirectory(
        _ url: URL
    ) throws {
        let fd = url.path.withCString {
            open($0, O_RDONLY)
        }
        guard fd >= 0 else {
            throw SourceSnapshotError.invalidSource(
                "Cannot open snapshot directory"
            )
        }
        defer { _ = close(fd) }

        guard fsync(fd) == 0 else {
            throw SourceSnapshotError.invalidSource(
                "Cannot sync snapshot directory"
            )
        }
    }
}

private struct Digest256 {
    #if canImport(CryptoKit)
    private var hash = SHA256()

    mutating func update(_ data: Data) {
        hash.update(data: data)
    }

    mutating func finish() -> String {
        hash.finalize()
            .map { String(format: "%02x", $0) }
            .joined()
    }
    #else
    // Portable SHA-256 fallback for Linux test environments.
    private static let k: [UInt32] = [
        0x428a2f98,0x71374491,0xb5c0fbcf,0xe9b5dba5,
        0x3956c25b,0x59f111f1,0x923f82a4,0xab1c5ed5,
        0xd807aa98,0x12835b01,0x243185be,0x550c7dc3,
        0x72be5d74,0x80deb1fe,0x9bdc06a7,0xc19bf174,
        0xe49b69c1,0xefbe4786,0x0fc19dc6,0x240ca1cc,
        0x2de92c6f,0x4a7484aa,0x5cb0a9dc,0x76f988da,
        0x983e5152,0xa831c66d,0xb00327c8,0xbf597fc7,
        0xc6e00bf3,0xd5a79147,0x06ca6351,0x14292967,
        0x27b70a85,0x2e1b2138,0x4d2c6dfc,0x53380d13,
        0x650a7354,0x766a0abb,0x81c2c92e,0x92722c85,
        0xa2bfe8a1,0xa81a664b,0xc24b8b70,0xc76c51a3,
        0xd192e819,0xd6990624,0xf40e3585,0x106aa070,
        0x19a4c116,0x1e376c08,0x2748774c,0x34b0bcb5,
        0x391c0cb3,0x4ed8aa4a,0x5b9cca4f,0x682e6ff3,
        0x748f82ee,0x78a5636f,0x84c87814,0x8cc70208,
        0x90befffa,0xa4506ceb,0xbef9a3f7,0xc67178f2
    ]

    private var state: [UInt32] = [
        0x6a09e667,0xbb67ae85,0x3c6ef372,0xa54ff53a,
        0x510e527f,0x9b05688c,0x1f83d9ab,0x5be0cd19
    ]
    private var buffer: [UInt8] = []
    private var count: UInt64 = 0

    private static func rotate(
        _ value: UInt32,
        _ amount: UInt32
    ) -> UInt32 {
        (value >> amount) | (value << (32 - amount))
    }

    mutating func update(_ data: Data) {
        count += UInt64(data.count)

        for byte in data {
            buffer.append(byte)
            if buffer.count == 64 {
                process(buffer)
                buffer.removeAll(keepingCapacity: true)
            }
        }
    }

    private mutating func process(_ block: [UInt8]) {
        var w = [UInt32](repeating: 0, count: 64)

        for i in 0..<16 {
            let j = i * 4
            w[i] =
                UInt32(block[j]) << 24 |
                UInt32(block[j+1]) << 16 |
                UInt32(block[j+2]) << 8 |
                UInt32(block[j+3])
        }

        for i in 16..<64 {
            let s0 = Self.rotate(w[i-15], 7) ^
                     Self.rotate(w[i-15], 18) ^
                     (w[i-15] >> 3)
            let s1 = Self.rotate(w[i-2], 17) ^
                     Self.rotate(w[i-2], 19) ^
                     (w[i-2] >> 10)
            w[i] = w[i-16] &+ s0 &+ w[i-7] &+ s1
        }

        var a = state[0], b = state[1]
        var c = state[2], d = state[3]
        var e = state[4], f = state[5]
        var g = state[6], h = state[7]

        for i in 0..<64 {
            let s1 = Self.rotate(e, 6) ^
                     Self.rotate(e, 11) ^
                     Self.rotate(e, 25)
            let ch = (e & f) ^ ((~e) & g)
            let t1 = h &+ s1 &+ ch &+ Self.k[i] &+ w[i]
            let s0 = Self.rotate(a, 2) ^
                     Self.rotate(a, 13) ^
                     Self.rotate(a, 22)
            let maj = (a & b) ^ (a & c) ^ (b & c)
            let t2 = s0 &+ maj

            h = g; g = f; f = e; e = d &+ t1
            d = c; c = b; b = a; a = t1 &+ t2
        }

        let words = [a,b,c,d,e,f,g,h]
        for i in 0..<8 {
            state[i] = state[i] &+ words[i]
        }
    }

    mutating func finish() -> String {
        let bits = count &* 8
        buffer.append(0x80)

        while buffer.count % 64 != 56 {
            buffer.append(0)
        }

        for i in (0..<8).reversed() {
            buffer.append(
                UInt8(truncatingIfNeeded: bits >> UInt64(i * 8))
            )
        }

        while !buffer.isEmpty {
            process(Array(buffer.prefix(64)))
            buffer.removeFirst(64)
        }

        return state
            .map { String(format: "%08x", $0) }
            .joined()
    }
    #endif
}

#!/usr/bin/env python3
"""Validate S02 evidence consistency, not device-level acceptance."""
from __future__ import annotations

import argparse
import json
from datetime import datetime
from pathlib import Path

REQUIRED = {
    "s02_t1_finder_launch.json": {
        "testcase_id": "S02.T1",
        "requirement_id": "UI-16",
        "text": (
            "app_path",
            "bundled_contracts_path",
            "unicode_path_tested",
            "external_volume_path_tested",
        ),
        "true": (),
    },
    "s02_t2_bookmark.json": {
        "testcase_id": "S02.T2",
        "requirement_id": "UI-19",
        "text": ("error_message_offered",),
        "true": (
            "bookmark_created",
            "bookmark_resolved_on_reopen",
            "stale_or_revoked_detected",
            "repick_action_offered",
        ),
    },
}


def validate_file(path: Path) -> list[str]:
    errors: list[str] = []
    if path.name not in REQUIRED:
        return [f"{path}: unsupported evidence filename"]

    expected = REQUIRED[path.name]
    try:
        payload = json.loads(path.read_text(encoding="utf-8"))
    except (OSError, UnicodeError, ValueError) as exc:
        return [f"{path}: invalid evidence: {exc}"]

    if not isinstance(payload, dict):
        return [f"{path}: JSON root must be object"]

    for key in ("testcase_id", "requirement_id"):
        if payload.get(key) != expected[key]:
            errors.append(f"{path}: invalid {key}")

    if payload.get("status") != "PASS":
        errors.append(f"{path}: status is not PASS")

    for key in expected["text"]:
        value = payload.get(key)
        if not isinstance(value, str) or not value.strip():
            errors.append(f"{path}: missing {key}")

    for key in expected["true"]:
        if payload.get(key) is not True:
            errors.append(
                f"{path}: PASS invalid because {key} is not true"
            )

    timestamp = payload.get("timestamp")
    if not isinstance(timestamp, str):
        errors.append(f"{path}: missing timestamp")
    else:
        try:
            parsed = datetime.fromisoformat(
                timestamp.replace("Z", "+00:00")
            )
            if parsed.tzinfo is None:
                errors.append(f"{path}: timestamp missing offset")
        except ValueError:
            errors.append(f"{path}: invalid timestamp")

    return errors


def validate_directory(directory: Path) -> list[str]:
    return [
        error
        for filename in REQUIRED
        for error in validate_file(directory / filename)
    ]


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "evidence_dir",
        nargs="?",
        type=Path,
        default=Path("reports/evidence")
    )
    args = parser.parse_args()

    errors = validate_directory(args.evidence_dir)
    if errors:
        for error in errors:
            print("FAIL", error)
        return 1

    print("PASS S02 evidence consistency")
    print("Device-level acceptance remains separately required.")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
